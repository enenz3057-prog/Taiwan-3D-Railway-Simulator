# 首先，請確保您已經安裝了 OpenCV： # pip install opencv-python import cv2 # 定義圖片檔案的路徑 (假設圖片檔名為 image_5.png) image_path = 'image_5.png' # 載入圖片 # cv2.imread() 函數會讀取圖片，並將其作為一個 NumPy 陣列返回 img = cv2.imread(image_path) # 檢查圖片是否成功載入 if img is not None: # 顯示圖片視窗 # cv2.imshow() 函數會開啟一個視窗顯示圖片 cv2.imshow('Railway Track Image', img) # 等待使用者按下任意鍵 # cv2.waitKey(0) 會暫停程式執行，直到鍵盤事件發生 print("按任意鍵關閉圖片視窗...") cv2.waitKey(0) # 關閉所有開啟的 OpenCV 視窗 cv2.destroyAllWindows() else: print(f"無法載入圖片，請確認檔案路徑 '{image_path}' 是否正確。")
