// 8. 【臺鐵標準斜張懸臂精準擬真系統】 const poleMat = new THREE.MeshStandardMaterial({ color: 0xdddddd, roughness: 0.7 }); const armMat = new THREE.MeshStandardMaterial({ color: 0xcccccc, metalness: 0.6, roughness: 0.3 }); const insMat = new THREE.MeshStandardMaterial({ color: 0x5c2c16, roughness: 0.4 }); // 咖啡色絕緣子 const wireMat = new THREE.MeshBasicMaterial({ color: 0x111111 }); // 自動計算兩點間距離並生成圓柱桿件的輔助函式 function createCylinderBetween(p1, p2, radius, material) { const distance = p1.distanceTo(p2); const geometry = new THREE.CylinderGeometry(radius, radius, distance, 8); const mesh = new THREE.Mesh(geometry, material); const midpoint = new THREE.Vector3().addVectors(p1, p2).multiplyScalar(0.5); mesh.position.copy(midpoint); mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), new THREE.Vector3().subVectors(p2, p1).normalize()); return mesh; } const poleInterval = 15; let index = 0; for (let z = -75; z <= 75; z += poleInterval) { const group = new THREE.Group(); const side = (index % 2 === 0) ? 1 : -1; const poleX = 4.2 * side; const wireTargetX = 1.6 * side; // 主電桿 const pole = new THREE.Mesh(new THREE.CylinderGeometry(0.22, 0.25, 8.0, 16), poleMat); pole.position.set(poleX, 4.0, 0); pole.castShadow = true; group.add(pole); // 結構點位定義 const pPoleTop = new THREE.Vector3(poleX - (0.2 * side), 6.8, 0); // 電桿最頂端 const pPoleMid = new THREE.Vector3(poleX - (0.2 * side), 4.8, 0); // 電桿中段固定點 const pHorizontalEnd = new THREE.Vector3(wireTargetX, 6.8, 0); // 水平橫桿末端 // 1. 頂部水平橫桿 const topArm = createCylinderBetween(pPoleTop, pHorizontalEnd, 0.045, armMat); group.add(topArm); // 2. 上方斜撐小拉桿（從電桿頂端下方斜拉至水平橫桿中段） const pTopSlantStart = new THREE.Vector3(poleX - (0.2 * side), 6.1, 0); const pTopSlantEnd = new THREE.Vector3(poleX - (1.8 * side), 6.8, 0); const topSlantArm = createCylinderBetween(pTopSlantStart, pTopSlantEnd, 0.03, armMat); group.add(topSlantArm); // 3. 下方主斜張懸臂與絕緣子（從電桿中段斜向上拉至水平橫桿前端下方） const pInsStart = new THREE.Vector3(poleX - (1.4 * side), 5.5, 0); const pInsEnd = new THREE.Vector3(poleX - (2.1 * side), 5.9, 0); const pSlantEnd = new THREE.Vector3(wireTargetX - (0.1 * side), 6.4, 0); const slantPart1 = createCylinderBetween(pPoleMid, pInsStart, 0.035, armMat); const insulator = createCylinderBetween(pInsStart, pInsEnd, 0.065, insMat); // 咖啡色絕緣子 const slantPart2 = createCylinderBetween(pInsEnd, pSlantEnd, 0.035, armMat); group.add(slantPart1); group.add(insulator); group.add(slantPart2); // 4. 接觸線懸挂吊桿 const pContactDrop = new THREE.Vector3(wireTargetX, 5.2, 0); const dropArm = createCylinderBetween(pHorizontalEnd, pContactDrop, 0.025, armMat); group.add(dropArm); group.position.set(0, 0, z); scene.add(group); index++; }
