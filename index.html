using UnityEngine; public class TaiwanTrainController : MonoBehaviour { [Header("列車基本參數")] [Tooltip("最高時速 (km/h)")] public float maxSpeed = 130f; // 參考台鐵主流列車設定 [Tooltip("加速度 (km/h 每秒)")] public float acceleration = 8f; // 牽引動力 [Tooltip("一般煞車減速度")] public float brakingDeceleration = 15f;// 軔機煞車 [Tooltip("自然阻力")] public float naturalDrag = 2f; // 慣性阻力 [Header("目前狀態 (唯讀)")] [SerializeField] private float currentSpeedKmh = 0f; // 當前時速 void Update() { HandleControls(); ApplyMovement(); } // 1. 處理玩家輸入 (模擬駕駛桿與軔機) void HandleControls() { // 按住 W 鍵：推動主控桿（加速） if (Input.GetKey(KeyCode.W)) { currentSpeedKmh += acceleration * Time.deltaTime; } // 按住 S 鍵：操作軔機（煞車） else if (Input.GetKey(KeyCode.S)) { currentSpeedKmh -= brakingDeceleration * Time.deltaTime; } // 沒有按鍵時：模擬行進中的自然阻力與慣性減速 else { if (currentSpeedKmh > 0) { currentSpeedKmh -= naturalDrag * Time.deltaTime; if (currentSpeedKmh < 0) currentSpeedKmh = 0; } } // 限制速度不超過最高速、不倒退 currentSpeedKmh = Mathf.Clamp(currentSpeedKmh, 0f, maxSpeed); } // 2. 讓火車實際在地圖上移動 void ApplyMovement() { // 將時速 (km/h) 轉換為每秒移動的公尺數 (m/s) float speedMs = currentSpeedKmh / 3.6f; // 讓火車沿著自己的正前方 (Z軸) 移動 transform.Translate(Vector3.forward * speedMs * Time.deltaTime); } // 讓你在 Unity 畫面上即時看到目前時速 void OnGUI() { GUI.Box(new Rect(20, 20, 200, 60), "駕駛艙儀表板"); GUI.Label(new Rect(30, 45, 180, 20), $"目前時速: {currentSpeedKmh:F1} km/h"); } }
