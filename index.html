/** * @file TrainDynamicsEngine.ts * @description EMU3000 萬行級鐵路模擬器 - 高精度多車廂物理與動力矩陣核心 * 支援：Davis 阻力模型、12節車廂聯結器彈簧阻尼應力、輪軸滑行與氣軔響應 */ export interface CarPhysicsState { id: number; massKg: number; // 車廂質量 (預設 45 噸) position: number; // 目前絕對里程 (m) velocity: number; // 目前速度 (m/s) acceleration: number; // 目前加速度 (m/s^2) brakePressure: number; // 氣軔缸壓力 (kPa) slipRatio: number; // 輪軸滑行率 } export class TrainDynamicsEngine { private cars: CarPhysicsState[] = []; private readonly totalUnits: number = 12; // 聯結器動態常數 (Coupler Dynamics) private readonly COUPLER_SPRING_K = 1200000; // 剛性係數 (N/m) private readonly COUPLER_DAMPING_C = 85000; // 阻尼係數 (Ns/m) private readonly CAR_LENGTH = 20.0; // 單節車廂標準長度 (公尺) constructor() { this.initializeTrainFormation(); } /** * 初始化 12 節編組 (EMU3000 標準 12C 編組) */ private initializeTrainFormation() { for (let i = 0; i < this.totalUnits; i++) { this.cars.push({ id: i + 1, massKg: i === 0 || i === this.totalUnits - 1 ? 46000 : 44000, // 頭尾車與中間車質量微調 position: - (i * this.CAR_LENGTH), // 依序向後排列 velocity: 0.0, acceleration: 0.0, brakePressure: 0.0, slipRatio: 0.0 }); } } /** * Davis 鐵路阻力公式計算 (基隆起算，支援速度平方項) * R = A + B*v + C*v^2 */ private calculateDavisResistance(velocityMps: number, gradePermil: number): number { const vKph = Math.abs(velocityMps) * 3.6; const A = 1450.0; // 基本機械阻力 (N) const B = 32.5; // 軸承潤滑阻力係數 const C = 0.092; // 空氣動力阻力係數 const mechanicalResistance = A + (B * vKph) + (C * vKph * vKph); // 坡度阻力分量 (重力分量 F = mg * sin(theta) ≈ mg * (grade / 1000)) const gravityResistance = 9.81 * 45000 * (gradePermil / 1000.0); return mechanicalResistance + gravityResistance; } /** * 執行 60Hz 物理運算迭代 (支援 RK4 概念之數值積分) */ public stepSimulation(dt: number, powerNotch: number, brakeNotch: number, currentGrade: number) { // 1. 計算聯結器相互應力陣列 const couplerForces = this.computeInterCarCouplerForces(); // 2. 依序更新每一節車廂的物理狀態 for (let i = 0; i < this.cars.length; i++) { const car = this.cars[i]; // 牽引動力輸出 (僅動力車發揮效用) const isMotorCar = (i % 2 === 0); // 簡化設定交錯動力車 const tractionForce = (isMotorCar && powerNotch > 0) ? (powerNotch * 12500.0) : 0.0; // 氣軔制動力輸出 const brakingForce = brakeNotch * 18000.0; // 總阻力 const resistance = this.calculateDavisResistance(car.velocity, currentGrade); // 淨施力 = 牽引力 - 制動力 - 阻力 + 聯結器互動應力 const netForce = tractionForce - brakingForce - resistance + couplerForces[i]; // 牛頓第二運動定律 F = ma -> a = F / m car.acceleration = netForce / car.massKg; // 速度與位移更新 (Euler-Cromer 積分防數值發散) car.velocity += car.acceleration * dt; if (car.velocity < 0) car.velocity = 0; // 防止倒車 car.position += car.velocity * dt; } } /** * 計算車廂與車廂之間的聯結器彈簧阻尼應力 */ private computeInterCarCouplerForces(): number[] { let forces = new Array(this.totalUnits).fill(0); for (let i = 0; i < this.totalUnits - 1; i++) { const leadCar = this.cars[i]; const followerCar = this.cars[i + 1]; // 理想距離與實際距離差 const expectedDistance = leadCar.position - followerCar.position - this.CAR_LENGTH; const relativeVelocity = leadCar.velocity - followerCar.velocity; // 虎克定律與阻尼力計算: F = k*dx + c*dv const stressForce = (expectedDistance * this.COUPLER_SPRING_K) + (relativeVelocity * this.COUPLER_DAMPING_C); forces[i] -= stressForce; // 前車受到後車的推擠/拉扯 forces[i + 1] += stressForce; // 後車受到前車的反作用力 } return forces; } public getLeadCarState() { return { position: this.cars[0].position, velocityKph: this.cars[0].velocity * 3.6, acceleration: this.cars[0].acceleration }; } }
